function makeHouse(startY) {
  return { x: Math.random()*W, y: startY ?? H+60, s: Math.random()*22+10, vy: -(Math.random()*0.5+0.25), vx: (Math.random()-0.5)*0.3, ang: Math.random()*Math.PI*2, rot: (Math.random()-0.5)*0.012, alpha: Math.random()*0.45+0.15, col: Math.random()>0.5 ? '#c9a84c' : '#f0d080' };
}

function drawHouse(h) {
  const s = h.s;
  ctx.save(); ctx.translate(h.x, h.y); ctx.rotate(h.ang);
  ctx.globalAlpha = h.alpha; ctx.strokeStyle = h.col; ctx.lineWidth = 1.1;
  ctx.beginPath(); ctx.rect(-s*.55, -s*.4, s*1.1, s*.8); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(-s*.65,-s*.4); ctx.lineTo(0,-s*1.05); ctx.lineTo(s*.65,-s*.4); ctx.stroke();
  ctx.beginPath(); ctx.rect(-s*.12, s*.05, s*.24, s*.35); ctx.stroke();
  ctx.beginPath(); ctx.rect(-s*.42, -s*.2, s*.28, s*.26); ctx.stroke();
  ctx.restore();
}
